cmake_minimum_required(VERSION 3.24)
project(HaloWebTransport LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(HALO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
option(HALO_WEB_USE_MBEDTLS "Use the same TLS backend as the iOS experiment" ON)
option(HALO_WEB_BUILD_PROBES "Build local packet-delivery fixtures" ON)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR HALO_WEB_USE_MBEDTLS)
    set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(DISABLE_PACKAGE_CONFIG_AND_INSTALL OFF CACHE BOOL "" FORCE)
    set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
    set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
    set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/mbedtls_config.h" CACHE FILEPATH "" FORCE)
    add_subdirectory("${HALO_ROOT}/build/third_party/mbedtls" mbedtls EXCLUDE_FROM_ALL)
    if(NOT TARGET MbedTLS::MbedTLS)
        add_library(MbedTLS::MbedTLS ALIAS mbedtls)
    endif()
endif()
add_subdirectory("${HALO_ROOT}/build/third_party/libdatachannel" rtc EXCLUDE_FROM_ALL)
add_library(halo-web-transport STATIC frames.c socket_transport.c rtc_transport.cpp)
target_compile_options(halo-web-transport PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(halo-web-transport PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(halo-web-transport PUBLIC datachannel-static)
target_include_directories(halo-web-transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(HALO_WEB_BUILD_PROBES AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(halo-web-socket-probe tests/socket_probe.c)
    target_compile_options(halo-web-socket-probe PRIVATE -UNDEBUG)
    target_link_libraries(halo-web-socket-probe PRIVATE halo-web-transport)
    add_executable(halo-web-rtc-probe tests/rtc_probe.cpp)
    target_include_directories(halo-web-rtc-probe PRIVATE "${HALO_ROOT}/build/third_party/libdatachannel/deps/json/single_include")
    target_link_libraries(halo-web-rtc-probe PRIVATE halo-web-transport)
    if(APPLE)
        add_executable(halo-web-posix-probe tests/posix_probe.c "${HALO_ROOT}/port/ios/host/posix_net.c")
        target_compile_options(halo-web-posix-probe PRIVATE -UNDEBUG)
        target_compile_definitions(halo-web-posix-probe PRIVATE HALO_IOS=1 HALO_ENABLE_WEBRTC=1)
        target_include_directories(halo-web-posix-probe PRIVATE "${HALO_ROOT}/port/linux/src")
        target_link_libraries(halo-web-posix-probe PRIVATE halo-web-transport)
    endif()
elseif(HALO_WEB_BUILD_PROBES)
    enable_language(OBJC)
    add_executable(HaloWebProbe MACOSX_BUNDLE tests/ios_probe.m "${HALO_ROOT}/port/ios/host/posix_net.c")
    target_compile_definitions(HaloWebProbe PRIVATE HALO_IOS=1 HALO_ENABLE_WEBRTC=1)
    target_include_directories(HaloWebProbe PRIVATE "${HALO_ROOT}/port/linux/src")
    target_compile_options(HaloWebProbe PRIVATE -fobjc-arc -Wno-deprecated-declarations)
    target_link_libraries(HaloWebProbe PRIVATE halo-web-transport "-framework UIKit" "-framework Foundation")
    set_target_properties(HaloWebProbe PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/tests/Info.plist"
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.haloce.webtransportprobe
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
        XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
endif()
